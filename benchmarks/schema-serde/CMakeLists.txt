cmake_minimum_required(VERSION 3.17)
project(serde_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(SCHEMA_SERDE "Benchmark the schema-serde build" OFF)
find_package(AWSSDK REQUIRED COMPONENTS dynamodb)
add_executable(serde-bench bench.cpp)
target_link_libraries(serde-bench ${AWSSDK_LINK_LIBRARIES})
if (SCHEMA_SERDE)
  target_compile_definitions(serde-bench PRIVATE SCHEMA_SERDE)
endif()
